/* ============ FILTER ============ */
function currentWords() {

  return allWords.filter(w => w.cat === category);

/* ============ RENDER ============ */
function render() {
  flipped = false;
  answered = false;
  feedbackEl.textContent = "";
  feedbackEl.className = "feedback";

  const words = currentWords();
  if (!words.length) {
    cardEl.innerHTML = `<div class="prompt-label">empty</div><div class="meaning">no words in this category</div>`;
    progressEl.textContent = "0 / 0";
    return;
  }
  if (index >= words.length) index = 0;

  const w = words[index];
  progressEl.textContent = `${index + 1} / ${words.length}`;

  const showBack = () => `
    <div class="prompt-label">answer</div>
    <div class="hanzi ${w.hanzi.length > 2 ? "small" : ""}">${w.hanzi}</div>
    <div class="pinyin">${w.pinyin}</div>
    <div class="meaning">${w.meaning}</div>
  `;

  if (mode === "hanzi2meaning") {
    cardEl.innerHTML = `
      <div class="prompt-label">what does it mean?</div>
      <div class="hanzi ${w.hanzi.length > 2 ? "small" : ""}">${w.hanzi}</div>
      <div class="hint-flip">click to reveal</div>
    `;
    cardEl.onclick = () => {
      if (!flipped) { flipped = true; cardEl.innerHTML = showBack(); }
    };
  }

  if (mode === "meaning2hanzi") {
    cardEl.innerHTML = `
      <div class="prompt-label">which character?</div>
      <div class="meaning" style="font-size:26px;">${w.meaning}</div>
      <div class="hint-flip">click to reveal</div>
    `;
    cardEl.onclick = () => {
      if (!flipped) { flipped = true; cardEl.innerHTML = showBack(); }
    };
  }

  if (mode === "choice") {
    const others = words.filter(x => x.hanzi !== w.hanzi)
                        .sort(() => Math.random() - 0.5)
                        .slice(0, 3);
    const options = [w, ...others].sort(() => Math.random() - 0.5);

    cardEl.innerHTML = `
      <div class="prompt-label">pick the meaning</div>
      <div class="hanzi ${w.hanzi.length > 2 ? "small" : ""}">${w.hanzi}</div>
      <div class="options" id="options">
        ${options.map(o => `<button class="option" data-hanzi="${o.hanzi}">${o.meaning}</button>`).join("")}
      </div>
    `;
    cardEl.onclick = null;

    document.querySelectorAll(".option").forEach(btn => {
      btn.onclick = () => {
        if (answered) return;
        answered = true;
        const chosen = btn.dataset.hanzi;
        if (chosen === w.hanzi) {
          btn.classList.add("correct");
          feedbackEl.textContent = "✅ correct!";
          feedbackEl.className = "feedback ok";
        } else {
          btn.classList.add("wrong");
          document.querySelectorAll(".option").forEach(b => {
            if (b.dataset.hanzi === w.hanzi) b.classList.add("correct");
          });
          feedbackEl.textContent = `❌ correct: ${w.pinyin} — ${w.meaning}`;
          feedbackEl.className = "feedback no";
        }
      };
    });
  }
}

/* ============ NAVIGATION ============ */
function next() {
  const words = currentWords();
  index = (index + 1) % words.length;
  render();
}
function prev() {
  const words = currentWords();
  index = (index - 1 + words.length) % words.length;
  render();
}

document.addEventListener("keydown", (e) => {
  if (document.getElementById("cards-view").classList.contains("hidden")) return;
  if (e.key === "ArrowRight") next();
  if (e.key === "ArrowLeft") prev();
  if (e.key === " ") { e.preventDefault(); cardEl.click(); }
});

  const day = Math.floor(Date.now() / 86400000);
  return day % len;

function setGreeting() {
  const h = new Date().getHours();
  let cn, en;
  if (h < 6)       { cn = "还没睡吗？"; en = "still awake?"; }
  else if (h < 12) { cn = "早上好！";   en = "good morning!"; }
  else if (h < 18) { cn = "下午好！";   en = "good afternoon!"; }
  else             { cn = "晚上好！";   en = "good evening!"; }
  document.getElementById("greet-cn").textContent = cn;
  document.getElementById("greet-en").textContent = en;
}
